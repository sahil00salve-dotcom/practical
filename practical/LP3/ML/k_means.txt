
# EXPERIMENT 6
# K-Means and Hierarchical Clustering
# Sales Dataset

# STEP 1: Import libraries

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans, AgglomerativeClustering
from scipy.cluster.hierarchy import dendrogram, linkage


# STEP 2: Load dataset

df = pd.read_csv('/content/sales_data_sample.csv',
                 encoding='latin1')

print("First 5 rows:")
print(df.head())


# STEP 3: Check dataset

print("\nDataset Shape:")
print(df.shape)

print("\nDataset Information:")
print(df.info())

print("\nMissing Values:")
print(df.isnull().sum())


# STEP 4: Select numerical columns

X = df[['SALES', 'QUANTITYORDERED', 'PRICEEACH']]

X = X.dropna()


# STEP 5: Feature Scaling

scaler = StandardScaler()

X_scaled = scaler.fit_transform(X)


# STEP 6: Elbow Method

wcss = []

for i in range(1, 11):

    kmeans = KMeans(n_clusters=i,
                    random_state=42,
                    n_init=10)

    kmeans.fit(X_scaled)

    wcss.append(kmeans.inertia_)

plt.plot(range(1, 11), wcss, marker='o')

plt.title('Elbow Method')

plt.xlabel('Number of Clusters')

plt.ylabel('WCSS')

plt.show()


# STEP 7: K-Means Clustering

# Choose K based on elbow graph

kmeans = KMeans(n_clusters=3,
                random_state=42,
                n_init=10)

y_kmeans = kmeans.fit_predict(X_scaled)

df = df.loc[X.index].copy()

df['Cluster'] = y_kmeans

print("\nK-Means Clustering Results:")

print(df.head())


# STEP 8: Visualize K-Means

plt.figure(figsize=(8, 5))

sns.scatterplot(x=X['SALES'],
                y=X['QUANTITYORDERED'],
                hue=y_kmeans,
                palette='viridis')

plt.title('K-Means Clustering')

plt.xlabel('Sales')

plt.ylabel('Quantity Ordered')

plt.show()


# STEP 9: Hierarchical Clustering

hc = AgglomerativeClustering(n_clusters=3)

y_hc = hc.fit_predict(X_scaled)

df['Hierarchical_Cluster'] = y_hc

print("\nHierarchical Clustering Results:")

print(df.head())


# STEP 10: Draw Dendrogram

Z = linkage(X_scaled[:100], method='ward')

plt.figure(figsize=(10, 5))

dendrogram(Z)

plt.title('Hierarchical Clustering Dendrogram')

plt.xlabel('Data Points')

plt.ylabel('Distance')

plt.show()


# STEP 11: Display final results

print("\nK-Means Cluster Count:")

print(df['Cluster'].value_counts())

print("\nHierarchical Cluster Count:")

print(df['Hierarchical_Cluster'].value_counts())


# STEP 12: Cluster Analysis

print("\nAverage values of K-Means Clusters:")

print(df.groupby('Cluster')[
    ['SALES', 'QUANTITYORDERED', 'PRICEEACH']
].mean())
