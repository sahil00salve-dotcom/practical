
# Practical 4: Gradient Descent Algorithm

# Step 1: Import libraries
import numpy as np
import matplotlib.pyplot as plt

# Step 2: Define the function
def function(x):
    return (x + 3)**2

# Step 3: Define derivative
def derivative(x):
    return 2 * (x + 3)

# Step 4: Initialize values
x = 2
learning_rate = 0.1
iterations = 50

# Step 5: Apply Gradient Descent
for i in range(iterations):
    x = x - learning_rate * derivative(x)

# Step 6: Display result
print("Local Minima:", x)
print("Minimum Value:", function(x))

# Step 7: Plot graph
x_values = np.linspace(-6, 4, 100)
y_values = function(x_values)

plt.plot(x_values, y_values)
plt.scatter(x, function(x), color='red')

plt.xlabel("X")
plt.ylabel("Y")
plt.title("Gradient Descent")
plt.legend()
plt.grid()
plt.show()
